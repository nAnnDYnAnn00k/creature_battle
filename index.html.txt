      const wave2 = Math.sin(x * 0.5 - z * 0.35) * 0.25;

    function initiateMoveSequence(attacker, defender) {
      const isPower = attacker.energy >= 3;
      const move = isPower ? attacker.powerMove : attacker.moves[Math.floor(Math.random() * attacker.moves.length)];

        startPos: attacker.mesh.position.clone(),
        progress: 0, duration: isPower ? 2.45 : 1.65, impactTriggered: false
      };
    }

    function resolveMoveImpact(attacker, defender, move, isPower) {
      const guardMult = defender.guardTimer > 0 ? 0.5 : 1.0;
      const isCrit = Math.random() < (isPower ? attacker.crit + 0.2 : attacker.crit);
      const mitigation = 100 / (100 + defender.def * 3.2);
      
      const dmg = Math.max(2, Math.round(attacker.atk * move.mult * mitigation * (isCrit ? 1.5 : 1.0) * guardMult));
      defender.hp = Math.max(0, defender.hp - dmg);
      defender.hitFlash = isPower ? 0.35 : 0.20;

      const anim = move.anim;
      if (anim === 'horn_toss' || anim === 'power_suplex') {
        defender.victimState.liftY = isPower ? 0.75 : 0.45; defender.victimState.tiltZ = isPower ? Math.PI * 0.85 : 0.65;
      } else if (anim === 'power_cavitation' || anim === 'club_punch') {
        defender.victimState.knockX = isPower ? 0.95 : 0.45;
      } else {
        defender.victimState.knockX = isPower ? 0.6 : 0.3; defender.victimState.tiltZ = 0.3;
      }

      if (activeDmgPopups < MAX_POPUPS && (armySize < 10 || isPower || isCrit)) {
        spawnFloatingDamage(defender.mesh.position, dmg, isCrit, defender.side, isPower ? move.name : "");
      }

      if (defender.hp <= 0) {
        defender.dead = true;
        spawnHitSparks(defender.mesh.position, defender.color, 15);
      }
    }

      });
      if(count > 0) appendLog(`★ ARMY COMMAND: ${count} UNITS TRIGGERED OVERDRIVE!`, "text-fuchsia-300 font-extrabold");
    });

    const clock = new THREE.Clock();

    function animate() {
      requestAnimationFrame(animate);
      const rawDt = Math.min(clock.getDelta(), 0.1);
      const dt = rawDt * simSpeed;
      const elapsed = clock.getElapsedTime();

      let framesTillHUD = 0;

      if (battleState === 'FIGHTING') {
        for (const f of entities) {
          if (f.dead) continue;

          // Tick cooldowns
          if (f.guardTimer > 0) f.guardTimer -= dt;
          if (f.guardCooldown > 0) f.guardCooldown -= dt;

          const target = getClosestEnemy(f);
          if (target && !f.activeAction) {
            let dx = target.mesh.position.x - f.homeX;
            let dz = target.mesh.position.z - f.homeZ;
            const dist = Math.hypot(dx, dz);

            if (dist > 0.8) {
              // Boids Separation Logic - prevent armies from stacking perfectly
              for (const ally of entities) {
                if (ally !== f && ally.side === f.side && !ally.dead) {
                  const adx = f.homeX - ally.homeX;
                  const adz = f.homeZ - ally.homeZ;
                  const adist = Math.hypot(adx, adz);
                  if (adist > 0.001 && adist < 0.6) {
                    dx += (adx / adist) * 0.8; dz += (adz / adist) * 0.8;
                  }
                }
              }

              const mag = Math.hypot(dx, dz);
              if (mag > 0) {
                const speed = 2.8 * dt * f.spd;
                f.homeX += (dx / mag) * speed;
                f.homeZ += (dz / mag) * speed;
              }
            }

            if (dist < 1.95) f.timer -= dt;
            if (f.timer <= 0) {
              f.timer = f.cooldown * (0.8 + Math.random() * 0.4);
              initiateMoveSequence(f, target);
            }
          }

          // Visuals update (Locomotion & Action Animation)
          const ud = f.mesh.userData;
          if (f.activeAction) {
            const act = f.activeAction;
            act.progress += dt / act.duration;
            const p = Math.min(1, act.progress);
            const strike = p >= 0.18 ? Math.sin(((p - 0.18) / 0.82) * Math.PI) : 0;
            
            const defTarget = entities.find(e => e.id === act.defenderId);
            const tgX = defTarget ? defTarget.mesh.position.x : f.homeX + 1;
            const tgZ = defTarget ? defTarget.mesh.position.z : f.homeZ;

            f.mesh.position.x = THREE.MathUtils.lerp(act.startPos.x, tgX, strike * 0.5);
            f.mesh.position.z = THREE.MathUtils.lerp(act.startPos.z, tgZ, strike * 0.5);

            if (p >= 0.5 && !act.impactTriggered) {
              act.impactTriggered = true;
              if (defTarget && !defTarget.dead) resolveMoveImpact(f, defTarget, act.move, act.isPower);
            }
            if (p >= 1) f.activeAction = null;
          } else {
            f.mesh.position.x = THREE.MathUtils.lerp(f.mesh.position.x, f.homeX, dt * 8);
            f.mesh.position.z = THREE.MathUtils.lerp(f.mesh.position.z, f.homeZ, dt * 8);
          }

          f.mesh.position.y = THREE.MathUtils.lerp(f.mesh.position.y, getTerrainHeight(f.mesh.position.x, f.mesh.position.z), dt * 10);

          if (target && !f.activeAction) {
            const lookYaw = Math.atan2(-(target.mesh.position.z - f.mesh.position.z), target.mesh.position.x - f.mesh.position.x);
            f.mesh.rotation.y = THREE.MathUtils.lerp(f.mesh.rotation.y, lookYaw, dt * 5);
          }

          if (f.hitFlash > 0) {
            f.hitFlash -= dt; ud.mat.color.setHex(0xffffff);
          } else {
            ud.mat.color.setHex(f.color);
          }

          // Simple leg wiggle
          const moveSpeed = Math.abs(f.mesh.position.x - f.homeX) + Math.abs(f.mesh.position.z - f.homeZ);
          ud.legs.forEach(l => {
             l.mesh.rotation.z = Math.sin(elapsed * 8 + l.phase) * Math.min(0.15, moveSpeed * 5);
          });
        }

        framesTillHUD++;
        if (framesTillHUD > 5) { updateArmyHUD(); framesTillHUD = 0; }
      } else {
        // Death sinking animation
        for (const f of entities) {
          if (f.dead) {
            f.mesh.position.y -= dt * 0.2;
            f.mesh.userData.mat.opacity *= 0.95;
          }
        }
      }

      // Camera Army Tracking
      let sumX = 0, sumZ = 0, count = 0;
      for (const e of entities) {
        if (!e.dead) { sumX += e.mesh.position.x; sumZ += e.mesh.position.z; count++; }
      }
      const midX = count > 0 ? sumX / count : 0;
      const midZ = count > 0 ? sumZ / count : 0;
      
      const camTarget = new THREE.Vector3(midX, 0.5, midZ);
      const camYaw = elapsed * 0.05;
      const camRadius = 24.0 + (armySize * 0.2); // Zoom out based on army size

      camera.position.x = THREE.MathUtils.lerp(camera.position.x, camTarget.x + Math.sin(camYaw) * camRadius, rawDt * 2);
      camera.position.z = THREE.MathUtils.lerp(camera.position.z, camTarget.z + Math.cos(camYaw) * camRadius, rawDt * 2);
      camera.position.y = THREE.MathUtils.lerp(camera.position.y, 10.0 + (armySize * 0.15), rawDt * 2);
      camera.lookAt(camTarget);

      renderer.render(scene, camera);
    }

    window.onload = () => {
      rebuildWorldTopology();
      startBattle();
      animate();
    };
